### 设计决策

**为什么 KYC 只有 3 个工具，而不是覆盖全部 KycService 方法？**

| KycService 方法 | 文件 | AI 可用 | 原因 |
|-----------------|------|---------|------|
| `getMyKyc` | [`kyc.service.ts:186`](/apps/api/src/client/kyc/kyc.service.ts:186) | ✅ | 只读查询，无副作用的 |
| `getIdTypes` | [`kyc.service.ts:208`](/apps/api/src/client/kyc/kyc.service.ts:208) | ✅ | 只读查询，返回支持的证件类型 |
| `createSession` | [`kyc.service.ts:50`](/apps/api/src/client/kyc/kyc.service.ts:50) | ✅ | 创建活体检测会话（状态创建，AI 可触发） |
| `scanIdCard` | [`kyc.service.ts:242`](/apps/api/src/client/kyc/kyc.service.ts:242) | ❌ | 需要文件上传（Buffer），AI 无法处理 |
| `ocrIdCard` | [`kyc.service.ts:220`](/apps/api/src/client/kyc/kyc.service.ts:220) | ⚠️ | 需要先上传获取 storageKey，流程复杂 |
| `submitKyc` | [`kyc.service.ts:265`](/apps/api/src/client/kyc/kyc.service.ts:265) | ❌ | 需要上传身份证正反面照片 + 设备信息 |

**AI 的 KYC 角色是「引导」而非「完成」**：AI 查询状态、告知流程、提供 App 跳转按钮，用户点击后在 App 内完成证件上传和活体检测。

### Step 1: 创建 KYC Tools 工厂函数

**文件**: `apps/api/src/client/ai-customer/graph/tools/kyc-tools.ts`

```typescript
import { tool } from '@langchain/core/tools';
import { z } from 'zod';
import { KycService } from '@api/client/kyc/kyc.service';

/**
 * KYC 工具 — 工厂函数
 *
 * KYC 实名认证相关工具。注意：AI 只能做查询和引导，
 * 真正的证件上传和提交需要在 App 内完成。
 *
 * @param kycService - KycService 实例
 * @returns LangChain tool 数组
 */
export function createKycTools(kycService: KycService) {
  /**
   * Tool 1: 查询实名认证状态
   *
   * 返回当前用户的 KYC 实名认证状态，包括：
   * - kycStatus: 0=未开始/1=审核中/2=已驳回/3=需补充/4=已通过
   * - realName: 已认证的真实姓名（如果已通过）
   * - idNumber: 已认证的身份证号（已脱敏）
   * - rejectReason: 驳回原因（如果被驳回）
   */
  const kycStatusTool = tool(
    async ({ userId }) => {
      try {
        const result = await kycService.getMyKyc(userId);
        return JSON.stringify(result);
      } catch (error: any) {
        return `查询 KYC 状态失败：${error.message}`;
      }
    },
    {
      name: 'kyc_status',
      description:
        '查询当前登录用户的实名认证KYC状态。返回认证状态码、真实姓名、证件号、驳回原因等。' +
        '注意：只能查询当前登录用户，userId 由系统自动传入。',
      schema: z.object({
        userId: z.string().describe('用户ID（由系统自动传入）'),
      }),
    },
  );

  /**
   * Tool 2: 查询支持的证件类型
   *
   * 返回系统支持的实名认证证件类型列表。
   * 每个证件类型包含 id、name（名称）、status（状态）等信息。
   */
  const idTypesTool = tool(
    async () => {
      try {
        const result = await kycService.getIdTypes();
        return JSON.stringify(result);
      } catch (error: any) {
        return `查询证件类型失败：${error.message}`;
      }
    },
    {
      name: 'kyc_id_types',
      description:
        '查询系统支持的实名认证证件类型列表。例如身份证、护照等。' +
        '不需要用户ID，返回所有支持的证件类型。',
      schema: z.object({
        // 无需参数，保留空对象以符合 tool 接口
      }),
    },
  );

  /**
   * Tool 3: 创建活体检测会话
   *
   * 【状态创建操作】为当前用户创建一个活体检测会话。
   * 活体检测是 KYC 实名认证的其中一步，用于验证用户是真人。
   *
   * 注意：创建会话后，用户需要在 App 内完成活体检测操作。
   */
  const createSessionTool = tool(
    async ({ userId }) => {
      try {
        const result = await kycService.createSession(userId);
        return JSON.stringify(result);
      } catch (error: any) {
        return `创建活体检测会话失败：${error.message}`;
      }
    },
    {
      name: 'kyc_create_session',
      description:
        '【状态创建操作】为当前登录用户创建一个活体检测Liveness会话。' +
        '执行此操作后，用户需要在 App 内通过活体检测来完成验证。' +
        'userId 由系统自动传入，不需要用户提供。',
      schema: z.object({
        userId: z.string().describe('用户ID（由系统自动传入）'),
      }),
    },
  );

  return [kycStatusTool, idTypesTool, createSessionTool];
}
```

### Step 2: 创建 KYC Agent 节点

**文件**: `apps/api/src/client/ai-customer/graph/nodes/kyc-agent.node.ts`

```typescript
import { createReactAgent } from '@langchain/langgraph/prebuilt';
import { ChatOllama } from '@langchain/ollama';
import { KycService } from '@api/client/kyc/kyc.service';
import { createKycTools } from '../tools/kyc-tools';
import { AgentStateType } from '../state';

/**
 * 构建 KYC ReAct Agent
 *
 * 处理实名认证相关的查询：查询认证状态、证件类型、创建活体检测会话。
 * 注意：KYC 的完整流程（证件上传、提交审核）需要在 App 内完成，
 * AI 只做状态查询和流程引导，并提供 App 跳转按钮。
 */
function buildKycAgent(kycService: KycService) {
  return createReactAgent({
    llm: new ChatOllama({
      model: 'qwen3.5:0.8b',
      temperature: 0,
    }),
    tools: createKycTools(kycService),
    messageModifier:
      '你是 JoyMini 的实名认证助手。你的职责是帮助用户了解实名认证KYC状态和流程。\n\n' +
      '你可以：\n' +
      '1. 查询用户的实名认证状态并解释状态含义\n' +
      '2. 告知系统支持的证件类型\n' +
      '3. 为用户创建活体检测会话（引导用户在 App 内完成检测）\n\n' +
      '注意：\n' +
      '- KYC 状态 0=未开始 1=审核中 2=已驳回 3=需补充 4=已通过\n' +
      '- 如果用户要完成实名认证，请引导他们去 App 内的 KYC 页面\n' +
      '- 证件上传和提交审核需要在 App 内完成，AI 无法处理文件上传\n' +
      '- 提现操作需要先完成 KYC 实名认证',
  });

  return async (state: AgentStateType) => {
    const result = await agentApp.invoke({ messages: state.messages });
    return { messages: result.messages };
  };
}

/**
 * KYC Agent 节点工厂函数
 *
 * 在 AiCustomerService 中调用，传入 KycService 实例。
 *
 * @example
 * const kycAgentNode = createKycAgentNode(kycService);
 */
export function createKycAgentNode(kycService: KycService) {
  const agentApp = buildKycAgent(kycService);
  return async (state: AgentStateType) => {
    const result = await agentApp.invoke({ messages: state.messages });
    return { messages: result.messages };
  };
}
```

### Step 3: 更新 Wallet Tools — 追加 getKycStatus 工具

**为什么需要？** 用户提现需要先完成 KYC 实名认证。Wallet Agent 在处理提现前应该能检查用户的 KYC 状态。

**文件**: `apps/api/src/client/ai-customer/graph/tools/wallet-tools.ts`

在 `createWalletTools` 函数中追加第 4 个工具：

```typescript
import { KycService } from '@api/client/kyc/kyc.service';

export function createWalletTools(
  walletService: WalletService,
  clientWalletService: ClientWalletService,
  kycService: KycService,  // ← 新增参数
) {
  // ... 原有的 3 个工具保持不变 ...

  /**
   * Tool 4: 查询 KYC 实名认证状态
   *
   * 提现操作需要用户先完成实名认证。此工具用于检查当前用户的 KYC 状态。
   */
  const kycStatusTool = tool(
    async ({ userId }) => {
      try {
        const result = await kycService.getMyKyc(userId);
        return JSON.stringify(result);
      } catch (error: any) {
        return `查询 KYC 状态失败：${error.message}`;

      name: 'kyc_status',

        '查询当前登录用户的实名认证KYC状态。提现前应检查此状态，' +

        'userId 由系统自动传入。',
      schema: z.object({
        userId: z.string().describe('用户ID（由系统自动传入）'),
      }),
    },
  );

  return [balanceTool, transactionsTool, applyWithdrawTool, kycStatusTool];
  //                                                         ↑ 新增
```

## 📝 技术要点

1. **Tool 工厂函数模式**：与 06-11 一致，`createKycTools(kycService)` 接收 NestJS Service 实例，返回 `tool()` 数组
2. **KYC 状态码**：`0=未开始` `1=审核中` `2=已驳回` `3=需补充` `4=已通过`
3. **AI 的边界**：AI 不能上传文件、不能提交 KYC 审核 — 这些必须在 App 内完成
4. **按钮引导模式**：同 [20-Flutter-AI-人工融合方案.md](20-Flutter-AI-人工融合方案.md) 的 transfer 按钮 — AI 回复提供 App 深链接
